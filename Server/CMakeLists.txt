cmake_minimum_required(VERSION 3.16)
project(quiz_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED jsoncpp)
pkg_check_modules(LIBEVENT REQUIRED libevent)
pkg_check_modules(LIBEVENT_PTHREADS REQUIRED libevent_pthreads)
pkg_check_modules(MYSQL REQUIRED mysqlclient)
find_package(Threads REQUIRED)
find_package(spdlog REQUIRED)
add_executable(quiz_server main.cpp myserver.cpp db.cpp tcpserver.cpp tcpsocket.cpp thread.cpp)
target_include_directories(quiz_server PRIVATE ${JSONCPP_INCLUDE_DIRS} ${LIBEVENT_INCLUDE_DIRS} ${MYSQL_INCLUDE_DIRS})
target_link_libraries(quiz_server PRIVATE ${JSONCPP_LIBRARIES} ${LIBEVENT_LIBRARIES} ${LIBEVENT_PTHREADS_LIBRARIES} ${MYSQL_LIBRARIES} Threads::Threads spdlog::spdlog)
